  deploy:
    name: Deploy production from master only
    if: github.event_name != 'pull_request' && github.ref == 'refs/heads/master'
    needs: quality
    runs-on: [self-hosted, jy-server]
    steps:
      - uses: actions/checkout@v4
      - name: Check existing host media and secrets
        run: |
          for p in /var/www/metamuseum/data/metamuseum.env \
                   /var/www/metamuseum/data/static_splat \
                   /var/www/metamuseum/data/static_image \
                   /var/www/metamuseum/data/static_gltf/matisse_dance.glb; do
            if [ ! -e "$p" ]; then
              echo "::error::missing $p; run scripts/init_host_data.sh first"
              exit 1
            fi
          done
          mkdir -p /var/www/metamuseum/data/wall_media
      - name: Preserve rollback image
        run: |
          if docker image inspect metamuseum:latest >/dev/null 2>&1; then
            docker tag metamuseum:latest metamuseum:rollback
          fi
      - name: Build and start
        run: docker compose -f docker-compose.prod.yml up -d --build --remove-orphans
      - name: Health and actual page smoke
        run: |
          for i in $(seq 1 30); do
            if curl -fsS http://127.0.0.1:51736/health; then break; fi
            sleep 2
          done
          curl -fsS http://127.0.0.1:51736/health
          curl -fsS http://127.0.0.1:51736/ | grep -q 'id="root"'
      - name: Roll back
        if: failure()
        run: |
          docker compose -f docker-compose.prod.yml logs --tail=50 app
          if docker image inspect metamuseum:rollback >/dev/null 2>&1; then
            docker tag metamuseum:rollback metamuseum:latest
            docker compose -f docker-compose.prod.yml up -d --no-build --remove-orphans
          fi
      - name: Cap image and build cache
        if: success()
        run: |
          docker image prune -f
          docker builder prune -f --max-used-space 5GB
